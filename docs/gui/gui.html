  40def _app_base_dir():
  41    """Return the application directory holding the executable and ``config/``.
  42
  43    In onefile mode the extracted executable lives in a temporary directory
  44    that disappears on exit, so the original binary directory is used instead
  45    to keep projects and configuration persistent.
  46
  47    Returns:
  48        str: Absolute path to the application base directory.
  49    """
  50    if _ONEFILE:
  51        containing = getattr(_compiled_info, "containing_dir", None)
  52        if containing:
  53            return os.path.abspath(containing)
  54        onefile_dir = os.environ.get("NUITKA_ONEFILE_DIRECTORY")
  55        if onefile_dir:
  56            return os.path.abspath(onefile_dir)
  57    if _COMPILED:
  58        return os.path.dirname(os.path.abspath(sys.executable))
  59    return os.path.dirname(os.path.abspath(__file__))

  62def _app_executable():
  63    """Return the executable path used to relaunch the app or run workers.
  64
  65    Returns ``None`` when the app runs from source (not compiled); in that
  66    case the worker must be started with the current Python interpreter.
  67    """
  68    if _ONEFILE:
  69        # sys.argv[0] points to the original onefile binary (not the temporary one).
  70        executable = os.path.abspath(sys.argv[0])
  71        if os.path.isfile(executable):
  72            return executable
  73        containing = getattr(_compiled_info, "containing_dir", None)
  74        if containing:
  75            return os.path.join(os.path.abspath(containing), os.path.basename(sys.argv[0]))
  76    if _COMPILED:
  77        return os.path.abspath(sys.executable)
  78    return None

 124def create_desktop_entry():
 125    """Register an application shortcut in the system menu.
 126
 127    The ``.desktop`` file needs an "Exec" entry pointing to a program that is
 128    really executable. In compiled mode the binary itself is used; during
 129    development the current Python interpreter plus ``gui.py`` is used (the
 130    script alone is not executable and the desktop environment would discard
 131    the entry).
 132
 133    The icon is copied to a persistent location because, in onefile mode, the
 134    packaged PNG exists only inside the temporary extraction directory, which
 135    is deleted when the application closes.
 136    """
 137    try:
 138        # Use XDG_DATA_HOME when defined (that is exactly where system menus
 139        # look for entries); otherwise fall back to the user default location.
 140        xdg_data_home = os.environ.get("XDG_DATA_HOME")
 141        if xdg_data_home:
 142            data_dir = pathlib.Path(xdg_data_home)
 143        else:
 144            data_dir = pathlib.Path.home() / ".local" / "share"
 145        desktop_dir = data_dir / "applications"
 146        desktop_dir.mkdir(parents=True, exist_ok=True)
 147        desktop_file = desktop_dir / "multiflora.desktop"
 148
 149        executable = _app_executable()
 150        if _COMPILED and executable:
 151            exec_args = [executable]
 152        else:
 153            exec_args = [sys.executable, os.path.abspath(__file__)]
 154        exec_value = " ".join(_desktop_quote(arg) for arg in exec_args)
 155
 156        icon_value = ""
 157        icon_source = pathlib.Path(__file__).resolve().parent / "assets" / "multiflora.png"
 158        if not icon_source.is_file():
 159            icon_source = pathlib.Path(_BASE_DIR) / "assets" / "multiflora.png"
 160        if icon_source.is_file():
 161            icon_dir = data_dir / "multiflora"
 162            icon_dir.mkdir(parents=True, exist_ok=True)
 163            icon_dest = icon_dir / "multiflora.png"
 164            try:
 165                shutil.copy2(icon_source, icon_dest)
 166                icon_value = str(icon_dest)
 167            except Exception:
 168                # In onefile the source path lives in the temporary extraction
 169                # directory and stops existing when the app closes; in that
 170                # case it is better to omit the icon than to point to a
 171                # volatile file.
 172                icon_value = "" if _ONEFILE else str(icon_source)
 173
 174        desktop_content = (
 175            "[Desktop Entry]\n"
 176            "Version=1.0\n"
 177            "Type=Application\n"
 178            "Name=Multiflora\n"
 179            "Comment=Platform for Automation and Deployment of Networks\n"
 180            f"Exec={exec_value}\n"
 181        )
 182        if icon_value:
 183            desktop_content += f"Icon={icon_value}\n"
 184        desktop_content += "Terminal=false\nCategories=Utility;\n"
 185
 186        # Rewrite only when the content changed, fixing old or invalid entries
 187        # instead of keeping them forever.
 188        if desktop_file.is_file():
 189            try:
 190                if desktop_file.read_text(encoding="utf-8") == desktop_content:
 191                    return
 192            except Exception:
 193                pass
 194
 195        desktop_file.write_text(desktop_content, encoding="utf-8")
 196        desktop_file.chmod(0o644)
 197
 198        updater = shutil.which("update-desktop-database")
 199        if updater:
 200            subprocess.run(
 201                [updater, str(desktop_dir)],
 202                check=False,
 203                stdout=subprocess.DEVNULL,
 204                stderr=subprocess.DEVNULL,
 205            )
 206    except Exception as exc:
 207        # Failures here must not go unnoticed: console logging is not set up
 208        # yet at this point, so write straight to stderr to make the error
 209        # visible in any terminal execution.
 210        message = f"Could not create desktop entry: {exc}"
 211        logging.warning(message)
 212        try:
 213            print(f"Multiflora: {message}", file=sys.stderr)
 214        except Exception:
 215            pass

 257def _is_cache_path(path):
 258    """Check whether a path points to one of the internal cache files.
 259
 260    The internal files are the session cache (``.last_project.json``) and the
 261    continuous auto-save cache (``.autosave_project.json``). They must never
 262    be treated as the user's project file.
 263
 264    Args:
 265        path (str): Path to test.
 266
 267    Returns:
 268        bool: ``True`` when the path matches either internal cache file.
 269    """
 270    return _same_path(path, _CACHE_FILE) or _same_path(path, _AUTOSAVE_FILE)

 272def _same_path(path_a, path_b):
 273    """Check whether two paths point to the same physical file or location.
 274
 275    When both paths exist the comparison uses ``os.path.samefile`` (which
 276    resolves symlinks); otherwise the paths are normalized, including case
 277    folding on case-insensitive filesystems.
 278
 279    Args:
 280        path_a (str): First file path.
 281        path_b (str): Second file path.
 282
 283    Returns:
 284        bool: ``True`` when both paths represent the same destination.
 285    """
 286    if not path_a or not path_b:
 287        return False
 288    try:
 289        path_a = os.path.abspath(path_a)
 290        path_b = os.path.abspath(path_b)
 291    except Exception:
 292        return False
 293
 294    try:
 295        if os.path.exists(path_a) and os.path.exists(path_b):
 296            return os.path.samefile(path_a, path_b)
 297    except Exception:
 298        pass
 299
 300    return os.path.normcase(os.path.normpath(path_a)) == os.path.normcase(os.path.normpath(path_b))

 336def _check_path_consistency(path, field_type, queue_dict, entry_widget):
 337    """Run the consistency checks on a path that is about to be accepted.
 338
 339    The checks cover: files of the same queue spread across different
 340    subdirectories, a probable inventory/data swap (judged by file name), the
 341    same file used twice in one queue (hard block) and a file already in use
 342    by another queue (warning). Applies both to dialog selections and to
 343    manually typed paths.
 344
 345    Args:
 346        path (str): Path to validate.
 347        field_type (str): One of ``inventory``, ``data`` or ``template``.
 348        queue_dict (dict): Queue widgets and state the path belongs to.
 349        entry_widget (tk.Entry): Entry being validated (excluded from the
 350            duplicate search, since it is the field receiving the path).
 351
 352    Returns:
 353        bool: ``True`` when the path is accepted (with or without warnings).
 354    """
 355    resolved = _resolve_config_path(path)
 356
 357    # 1) Files of the same queue pointing to different subdirectories
 358    try:
 359        current_subdir = os.path.dirname(os.path.relpath(resolved, _CONFIG_DIR))
 360        existing_subdirs = []
 361        for k in ["inventory", "data", "template"]:
 362            if k in queue_dict and queue_dict[k] != entry_widget:
 363                other_raw = queue_dict[k].get().strip()
 364                other_path = _resolve_config_path(other_raw)
 365                if other_raw and os.path.exists(other_path):
 366                    other_sub = os.path.dirname(os.path.relpath(other_path, _CONFIG_DIR))
 367                    if other_sub:
 368                        existing_subdirs.append((k, other_sub))
 369        if current_subdir and existing_subdirs:
 370            unique_existing = set(sub for _, sub in existing_subdirs)
 371            if current_subdir not in unique_existing:
 372                names_field = ", ".join([f"{c} ({s})" for c, s in existing_subdirs])
 373                msg = (f"The file is in '{current_subdir}', but other files in this queue "
 374                       f"are in different subdirectories:\n\n{names_field}\n\n"
 375                       f"Do you want to continue anyway?")
 376                if not messagebox.askyesno("Warning: Different Subdirectories", msg):
 377                    return False
 378    except Exception as e:
 379        logging.warning(f"Error in subdirectory validation: {e}")
 380
 381    # 2) Probable inventory x data swap (judged by file name)
 382    file_name = os.path.basename(resolved).lower()
 383    inversion = False
 384    msg = ""
 385    if field_type == "inventory" and "data" in file_name:
 386        inversion = True
 387        msg = f"Selected a file with 'data' in the name for the Inventory field.\n\nFile: {file_name}\n\nAre you sure?"
 388    elif field_type == "data" and ("inv" in file_name or "inventory" in file_name):
 389        inversion = True
 390        msg = f"Selected a file with 'inventory' (or inv) in the name for the Data field.\n\nFile: {file_name}\n\nAre you sure?"
 391    if inversion and not messagebox.askyesno("Warning: Possible Inversion", msg):
 392        return False
 393
 394    # 3) Same file used twice in the same queue (hard block)
 395    for k in ["inventory", "data", "template"]:
 396        if k in queue_dict and queue_dict[k] != entry_widget:
 397            other_raw = queue_dict[k].get().strip()
 398            if other_raw and _resolve_config_path(other_raw) == resolved:
 399                messagebox.showerror("Error: Duplicate File",
 400                    "You cannot use the same file in different fields of the same queue.")
 401                return False
 402
 403    # 4) File already used by another queue (warning)
 404    used_in_other_queues = []
 405    for q in _execution_queues:
 406        if q["id"] != queue_dict["id"]:
 407            for k in ["inventory", "data", "template"]:
 408                if k in q and _resolve_config_path(q[k].get().strip()) == resolved:
 409                    used_in_other_queues.append(q["title_label"].cget("text"))
 410                    break
 411    if used_in_other_queues:
 412        queue_names = "\n- ".join(used_in_other_queues)
 413        if not messagebox.askyesno("Warning: File in Use",
 414            f"This file is already being used in the following queues:\n\n- {queue_names}\n\nDo you want to use it anyway?"):
 415            return False
 416
 417    return True

 419def _on_entry_focusout(entry_widget, field_type, queue_dict):
 420    """Validate manually typed paths when the field loses focus.
 421
 422    Applies the same consistency warnings used by dialog selections.
 423
 424    Args:
 425        entry_widget (tk.Entry): Entry that lost focus.
 426        field_type (str): One of ``inventory``, ``data`` or ``template``.
 427        queue_dict (dict): Queue widgets and state the entry belongs to.
 428    """
 429    raw = entry_widget.get().strip()
 430    if not raw:
 431        return
 432    resolved = _resolve_config_path(raw)
 433    key = (field_type, os.path.normpath(resolved))
 434    done = queue_dict.setdefault("_validated_paths", set())
 435    if key in done:
 436        return
 437    if not os.path.isfile(resolved):
 438        return  # missing files are blocked later by get_queue_errors
 439    accepted = _check_path_consistency(resolved, field_type, queue_dict, entry_widget)
 440    done.add(key)
 441    if not accepted:
 442        entry_widget.select_range(0, tk.END)   # keep the text selected for correction
 443        entry_widget.icursor(tk.END)

 463def _resolve_config_path(path):
 464    """Normalize user-typed paths, accepting relative entries.
 465
 466    Absolute paths are only normalized; relative paths are resolved first
 467    against ``config/`` and, as a fallback, against the application base
 468    directory.
 469
 470    Args:
 471        path (str): Raw path (absolute or relative).
 472
 473    Returns:
 474        str: Resolved absolute path (or the ``config/`` candidate when the
 475        file does not exist in either location).
 476    """
 477    if not path:
 478        return path
 479    p = os.path.expanduser(path.strip())
 480    if os.path.isabs(p):
 481        return os.path.normpath(p)
 482    candidate = os.path.normpath(os.path.join(_CONFIG_DIR, p))
 483    if os.path.exists(candidate):
 484        return candidate
 485    candidate_base = os.path.normpath(os.path.join(_BASE_DIR, p))
 486    if os.path.exists(candidate_base):
 487        return candidate_base
 488    return candidate

 627def parse_device_line(queue_dict, line):
 628    """Parse one worker output line read from the console tab.
 629
 630    Two patterns are recognized: the inventory validation line (which creates
 631    the device node with its platform name) and status messages mentioning a
 632    registered host, classified through keyword lists (``SUCCESS_WORDS``,
 633    ``_FAIL_WORDS``, applying/creating/solving and connecting). Tree updates
 634    are scheduled on the Tk main thread through ``root.after``.
 635
 636    Args:
 637        queue_dict (dict): Queue the line belongs to.
 638        line (str): Raw line read from the worker output.
 639    """
 640    qid = queue_dict["id"]
 641    m = _RE_VALIDATING.search(line)
 642    if m:
 643        h, p = m.group("host"), m.group("platform")
 644        root.after(0, lambda h=h, p=p: sidebar_add_device(qid, h, p, "detected"))
 645        return
 646    registered_hosts = _device_status.get(qid, {})
 647
 648    if not registered_hosts:
 649        return
 650
 651    line_lower = line.lower()
 652    matched_host = None
 653
 654    for host in registered_hosts:
 655        if re.search(rf"\b{re.escape(host.lower())}\b", line_lower):
 656            matched_host = host
 657            break
 658
 659    if not matched_host:
 660        return
 661
 662    if any(w in line_lower for w in SUCCESS_WORDS):
 663        status = "success"
 664    elif any(w in line_lower for w in _FAIL_WORDS):
 665        status = "failure"
 666    elif any(w in line_lower for w in ("applying", "creating", "solving")):
 667        status = "applying"
 668    elif "connecting" in line_lower:
 669        status = "connecting"
 670    else:
 671        return
 672
 673    root.after(0, lambda h=matched_host, s=status: sidebar_set_device_status(qid, h, s))

 698def select_file(entry_widget, title, filetypes, field_type, queue_dict):
 699    """Open a file dialog until the user picks an accepted path (or cancels).

 701    Every selected path goes through ``_check_path_consistency``; rejected

 703    ``entry_widget`` and its directory is remembered as the queue's last used
 704    directory.

 706    Args:

 711        queue_dict (dict): Queue the entry belongs to.
 712    """
 713    while True:
 714        start_dir = queue_dict.get("last_used_directory", _CONFIG_DIR)

 716            start_dir = _CONFIG_DIR
 717        path = filedialog.askopenfilename(title=title, filetypes=filetypes, initialdir=start_dir)
 718        if not path:
 719            return

 721            continue
 722        queue_dict["last_used_directory"] = os.path.dirname(path)
 723        queue_dict.setdefault("_validated_paths", set()).add((field_type, os.path.normpath(path)))
 724        entry_widget.delete(0, tk.END)
 725        entry_widget.insert(0, path)
 726        break

 825def get_queue_errors(queue_dict):
 826    """Collect the problems found in the path fields of a queue.
 827
 828    Checks for empty fields, files missing on disk and two fields pointing to
 829    the same file.
 830
 831    Args:
 832        queue_dict (dict): Queue whose fields are inspected.
 833
 834    Returns:
 835        list[str]: Human-readable error descriptions (empty when the queue is
 836        ready to run).
 837    """
 838    errors = []
 839    fields = {
 840        "Inventory path": queue_dict["inventory"].get().strip(),
 841        "Data path": queue_dict["data"].get().strip(),
 842        "Jinja Template path": queue_dict["template"].get().strip(),
 843    }
 844    for name_field, raw in fields.items():
 845        if not raw:
 846            errors.append(f"{name_field}: Empty")
 847        elif not os.path.isfile(_resolve_config_path(raw)):
 848            errors.append(f"{name_field}: File not found on disk")
 849    pairs = [(name, _resolve_config_path(raw)) for name, raw in fields.items() if raw]
 850    for i in range(len(pairs)):
 851        for j in range(i + 1, len(pairs)):
 852            if pairs[i][1] == pairs[j][1]:
 853                errors.append(f"{pairs[i][0]} and {pairs[j][0]} point to the same file")
 854    return errors

 856def execute_queue_logic(queue_dict, mode):
 857    """Run one queue in a separate worker process and relay its events.
 858
 859    The inventory must live inside a subdirectory of ``config/``; that
 860    subdirectory becomes the worker's configuration path. The worker (the
 861    compiled binary relaunched with ``--worker`` or the current interpreter
 862    running ``main.py``) is spawned as a subprocess, and every output line is
 863    forwarded to the queue logger and parsed to update the device tree.
 864
 865    Args:
 866        queue_dict (dict): Queue data and widgets to execute.
 867        mode (str): Execution mode (``dry-run`` or ``deploy``).
 868    """
 869    log = queue_dict["logger"]
 870    qid = queue_dict["id"]
 871    current_name = queue_dict["title_label"].cget("text")
 872    inv_path  = _resolve_config_path(queue_dict["inventory"].get().strip())
 873    data_path = _resolve_config_path(queue_dict["data"].get().strip())
 874    tpl_path  = _resolve_config_path(queue_dict["template"].get().strip())
 875
 876    try:
 877        rel_inv = os.path.relpath(inv_path, _CONFIG_DIR)
 878        father_dir = os.path.dirname(rel_inv)
 879        if not father_dir or father_dir == ".":
 880            log.error(f"File must be inside a subfolder of config/. Path: {inv_path}")
 881            return
 882    except ValueError as e:
 883        log.error(f"Path outside config/ directory: {e}")
 884        return
 885
 886    config_subdir = os.path.join(_CONFIG_DIR, father_dir)
 887    root.after(0, lambda: sidebar_clear_devices(qid))
 888
 889    # Build the worker command for a separate process:
 890    #  - compiled: relaunch the application executable with the "--worker" flag;
 891    #  - development: run main.py directly with the current interpreter.
 892    if _COMPILED:
 893        exe_path = _app_executable()
 894        if not exe_path or not os.path.isfile(exe_path):
 895            log.error(f"Could not locate the application executable to run {current_name}.")
 896            root.after(0, lambda: sidebar_finalize_queue(qid, mode))
 897            return
 898        cmd = [exe_path, "--worker", config_subdir, inv_path, data_path, tpl_path, current_name, f"--{mode}"]
 899    else:
 900        cmd = [sys.executable, "-u", main_script, config_subdir, inv_path, data_path, tpl_path, current_name, f"--{mode}"]
 901
 902    try:
 903        proc = subprocess.Popen(
 904            cmd,
 905            stdout=subprocess.PIPE,
 906            stderr=subprocess.STDOUT,
 907            text=True,
 908            errors="replace",
 909            bufsize=1,
 910            cwd=_BASE_DIR
 911        )
 912        for line in proc.stdout:
 913            line = line.rstrip()
 914            if line:
 915                log.info(line)
 916                parse_device_line(queue_dict, line)
 917        proc.wait()
 918    except Exception as e:
 919        log.error(f"Unexpected error while executing {current_name}: {e}")
 920
 921    root.after(0, lambda: sidebar_finalize_queue(qid, mode))

 923def run_single_queue(queue_dict, mode="dry-run"):
 924    """Validate and submit a single queue for background execution.
 925
 926    On validation errors an alert is shown and nothing runs. Otherwise the
 927    project is auto-saved and ``execute_queue_logic`` starts in a daemon
 928    thread, so the GUI never blocks.
 929
 930    Args:
 931        queue_dict (dict): Queue to execute.
 932        mode (str, optional): ``dry-run`` (default) or ``deploy``.
 933    """
 934    current_name = queue_dict["title_label"].cget("text")
 935    errors = get_queue_errors(queue_dict)
 936    
 937    if errors:
 938        msg_errors = "\n".join([f"- {err}" for err in errors])
 939        messagebox.showerror(
 940            "Validation Error",
 941            f"The '{current_name}' has issues:\n\n{msg_errors}\n\nPlease fix the paths before executing."
 942        )
 943        return
 944    
 945    auto_save_current_project()
 946    threading.Thread(
 947        target=execute_queue_logic,
 948        args=(queue_dict, mode),
 949        daemon=True,
 950        name=f"Queue-{queue_dict['id']}-{mode}"
 951    ).start()

1307def on_closing():
1308    """Handle the window close event.
1309
1310    With an official project open: "Yes" writes to the official file and syncs
1311    the caches; "No" overwrites the caches with the project content
1312    (effectively discarding the pending changes); "Cancel" aborts the exit.
1313    Without an official project, the work in progress is kept in the caches.
1314
1315    The Tk image reference is cleared before teardown to avoid errors during
1316    interpreter shutdown.
1317    """
1318    global icon  # Bring the image reference into scope
1319    icon = None  # Drop the image before Python shuts down
1320    if current_project_filepath and not _is_cache_path(current_project_filepath):
1321        if _has_unsaved_project_changes():
1322            response = messagebox.askyesnocancel(
1323                "Exit",
1324                "There are unsaved changes in the current project.\n"
1325                "Do you want to save them before exiting?"
1326            )
1327            if response is None:
1328                return
1329            if response:
1330                if not save_project_state(current_project_filepath):
1331                    return
1332                _sync_caches()
1333            else:
1334                _overwrite_caches_with_project()
1335        else:
1336            _sync_caches()
1337        root.quit()
1338        root.destroy()
1339    elif has_unsaved_changes():
1340        response = messagebox.askyesnocancel("Exit", "Do you want to save the current project before exiting?")
1341        if response is True:
1342            if save_project_as():
1343                _sync_caches()
1344                root.quit()
1345                root.destroy()
1346        elif response is False:
1347            _sync_caches()   # no official project: the draft survives for the next session
1348            root.destroy()
1349        else:
1350            return
1351    else:
1352        _sync_caches()
1353        root.quit()
1354        root.destroy()

1356def init_app():
1357    """Restore the session when the application opens.
1358
1359    Loads the session cache when present; if the load fails, a fresh queue is
1360    created. After a successful load, ``_offer_session_recovery`` checks
1361    whether the previous session ended abruptly and offers to recover the
1362    pending work.
1363    """
1364    if os.path.exists(_CACHE_FILE):
1365        if not load_project_state(_CACHE_FILE):
1366            add_queue()
1367        else:
1368            _offer_session_recovery()
1369    else:
1370        add_queue()

1372def _offer_session_recovery():
1373    """Offer recovery when the previous session did not end cleanly.
1374
1375    An unclean exit (crash, kill, power loss) is detected by comparing the
1376    continuous auto-save cache with the session cache: when they differ there
1377    were pending changes and the user decides whether to recover them.
1378    """
1379    if not os.path.exists(_AUTOSAVE_FILE):
1380        return
1381    try:
1382        with open(_AUTOSAVE_FILE, "r", encoding="utf-8") as f:
1383            autosave_state = json.load(f)
1384    except Exception:
1385        return
1386    if autosave_state == get_current_project_state():
1387        return
1388    response = messagebox.askyesno(
1389        "Session Recovery",
1390        "Unsaved changes from the previous session were detected.\n"
1391        "Do you want to recover them?"
1392    )
1393    if response:
1394        load_project_state(_AUTOSAVE_FILE) 
1395    else:
1396        save_project_state(_AUTOSAVE_FILE)

1472def _set_paned_split(top_height):
1473    """Apply explicit, complementary heights to both panes of ``right_paned``.
1474
1475    Because the two heights always add up to the available height of the
1476    PanedWindow, no orphan space (the gray band) is left after programmatic
1477    changes.
1478
1479    Args:
1480        top_height (int): Requested height of the queue pane; clamped to keep
1481            at least 100 px on the top and 150 px on the terminal pane.
1482    """
1483    right_paned.update_idletasks()
1484    total = right_paned.winfo_height()
1485    if total <= 1:
1486        return
1487    top_height = max(100, min(int(top_height), total - 150))
1488    right_paned.paneconfigure(top_frame, stretch="always")
1489    right_paned.paneconfigure(bottom_container, stretch="never")
1490    right_paned.paneconfigure(top_frame, height=top_height)
1491    right_paned.paneconfigure(bottom_container, height=total - top_height)

1493def _apply_default_paned_ratio():
1494    """Apply the initial split: compact terminal (~300 px), queues with the rest.

1496    Without this adjustment Tk hands the leftover space of the PanedWindow to
1497    its last pane (the terminal), which would fill almost the whole window at
1498    startup.
1499    """
1500    root.update_idletasks()
1501    total = right_paned.winfo_height()
1502    if total <= 1:
1503        return
1504    terminal_height = max(180, min(300, total - 250))
1505    _set_paned_split(total - terminal_height)
